B Method 2: Node.js Backend Host (Render / Railway / VPS) — Centralized Database

If you prefer a centralized backend server so all global users share the exact same backend database file on disk, deploy this lightweight Express server: 

const express = require('express');
const multer = require('multer');
const xlsx = require('xlsx');
const cors = require('cors');
const fs = require('fs');
const path = require('path');

const app = express();
const PORT = process.env.PORT || 3000;

app.use(cors());
app.use(express.json());
app.use(express.static('public')); // Serve index.html

const upload = multer({ dest: 'uploads/' });
const DB_FILE = path.join(__dirname, 'questions_db.json');

function readDB() {
  if (!fs.existsSync(DB_FILE)) return {};
  try { return JSON.parse(fs.readFileSync(DB_FILE, 'utf-8')); }
  catch (e) { return {}; }
}

function writeDB(data) {
  fs.writeFileSync(DB_FILE, JSON.stringify(data, null, 2));
}

// Upload endpoint for Admin
app.post('/api/upload', upload.single('questionBank'), (req, res) => {
  try {
    const workbook = xlsx.readFile(req.file.path);
    const sheet = workbook.Sheets[workbook.SheetNames[0]];
    const rawData = xlsx.utils.sheet_to_json(sheet, { header: 1 });
    
    // Auto-detect columns & group by section...
    // Saves parsed dataset to DB_FILE
    fs.unlinkSync(req.file.path);
    res.json({ success: true, message: "Uploaded & saved centrally." });
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

// Random 10-Question Test Endpoint (1 per section)
app.get('/api/test', (req, res) => {
  const db = readDB();
  const sections = Object.keys(db);
  const test = sections.map(sec => {
    const list = db[sec];
    return list[Math.floor(Math.random() * list.length)];
  });
  res.json({ questions: test });
});

app.listen(PORT, () => console.log(`Server running on port ${PORT}`));
